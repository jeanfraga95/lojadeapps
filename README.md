# lojadeapps
Loja de apps simples

Veja a estrutura da loja abaixo.
appstore/
├── app.py              # Backend Flask
├── database.py         # SQLite Setup
├── static/
│   ├── css/
│   │   └── style.css   # Design Moderno
│   └── js/
│       └── main.js     # Interatividade
├── templates/
│   ├── base.html
│   ├── login.html
│   ├── register.html
│   ├── dashboard.html
│   └── home.html
└── uploads/           # Pasta para APKs


instalação na maquina :
# 1. Crie a pasta
mkdir appstore && cd appstore

# 2. Instale dependências
pip install flask flask-login werkzeug

# 3. Crie todos os arquivos.
